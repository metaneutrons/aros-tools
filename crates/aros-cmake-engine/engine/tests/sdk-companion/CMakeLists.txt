cmake_minimum_required(VERSION 3.22)
project(SdkCompanionProbe NONE)
find_program(clang_driver NAMES clang REQUIRED)
file(REAL_PATH "${clang_driver}" CMAKE_C_COMPILER)
set(CMAKE_C_COMPILER_ID Clang)
set(CMAKE_C_COMPILER_TARGET x86_64-unknown-aros)
set(AROS_TARGET_TRIPLE x86_64-unknown-aros)
set(AROS_TARGET_CPU32 i386)
set(AROS_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/source")
set(AROS_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}")
set(AROS_TARGET_SYSROOT "${AROS_BUILD_DIR}/SYS/Developer")
set(AROS_GENINC_DIR "${AROS_BUILD_DIR}/gen/include")
set(AROS_SDK_INCLUDE_DIR "${AROS_TARGET_SYSROOT}/include")
file(MAKE_DIRECTORY "${AROS_SOURCE_DIR}" "${AROS_GENINC_DIR}" "${AROS_SDK_INCLUDE_DIR}/aros/stdc")
file(WRITE "${AROS_SOURCE_DIR}/probe.c" "int companion_probe(void) { return sizeof(void *); }\n")
# Policy double only. The compiler and emitted ELF are real; this fixture does
# not authenticate a source SDK or its original producing job.
include("${ENGINE_DIR}/NativeBuildContract.cmake")
include("${ENGINE_DIR}/SdkObjects.cmake")
function(aros_validate_native_consumer_compiler_target)
    if(NOT CMAKE_C_COMPILER_TARGET STREQUAL AROS_TARGET_TRIPLE)
        message(FATAL_ERROR "fixture primary compiler selection changed")
    endif()
endfunction()
function(_aros_native_source_selection_validated out)
    set(${out} TRUE PARENT_SCOPE)
endfunction()
set(role [=[{"selector_cpu":"i386","target":{"target_triple":"i386-unknown-aros","abi":{"elf_class":1,"elf_machine":3,"endian":"little"}},"compiler_options":["--target=i386-unknown-aros"],"library_dir":"lib32","required_runtime_library":"clang_rt.builtins-i386"}]=])
set(json "{\"companion_roles\":[${role}]}")
if(CASE STREQUAL "absent")
    set(json "{}")
elseif(CASE STREQUAL "wrong-class")
    string(REPLACE "\"elf_class\":1" "\"elf_class\":2" json "${json}")
elseif(CASE STREQUAL "wrong-machine")
    string(REPLACE "\"elf_machine\":3" "\"elf_machine\":62" json "${json}")
endif()
set_property(GLOBAL PROPERTY AROS_NATIVE_CONSUMER_VALIDATED_SCHEMA aros-sdk-consumer-validation-v1)
set_property(GLOBAL PROPERTY AROS_NATIVE_CONSUMER_BINDING_JSON "${json}")
set(output "${AROS_TARGET_SYSROOT}/lib32/probe.o")
if(CASE STREQUAL "wrong-library")
    set(output "${AROS_TARGET_SYSROOT}/lib/probe.o")
endif()
set(options -O2 -fno-common)
if(CASE STREQUAL "override")
    list(APPEND options -m32)
endif()
aros_compile_sdk_object(NAME companion SOURCE "${AROS_SOURCE_DIR}/probe.c"
    INTERMEDIATE "${AROS_BUILD_DIR}/gen/startup/32bit/probe.o"
    OUTPUT "${output}" LANGUAGE C COMPANION_CPU i386 OPTIONS ${options})
