cmake_minimum_required(VERSION 3.22)
project(SourceArchivesFixture C)
if(DEFINED FIXTURE_AR AND DEFINED FIXTURE_RANLIB)
    set(fixture_ar "${FIXTURE_AR}")
    set(fixture_ranlib "${FIXTURE_RANLIB}")
else()
    find_program(fixture_ar NAMES gar ar REQUIRED)
    find_program(fixture_ranlib NAMES granlib ranlib REQUIRED)
endif()
set(CMAKE_AR "${fixture_ar}")
set(CMAKE_RANLIB "${fixture_ranlib}")
foreach(tool CMAKE_C_COMPILER CMAKE_AR CMAKE_RANLIB)
    file(REAL_PATH "${${tool}}" selected)
    set(${tool} "${selected}")
endforeach()
set(AROS_BUILD_DIR "${CMAKE_BINARY_DIR}")
set(AROS_SOURCE_DIR "${CMAKE_BINARY_DIR}/source tree")
set(AROS_DEVELOPER_LIB_DIR "${CMAKE_BINARY_DIR}/SYS/Developer/lib")
set(AROS_NATIVE_BUILD_CONTRACT_VALIDATED ON)
set(AROS_TOOLCHAIN gnu)
file(MAKE_DIRECTORY "${AROS_SOURCE_DIR}")
file(COPY_FILE "${CMAKE_CURRENT_LIST_DIR}/one.c" "${AROS_SOURCE_DIR}/one.c" ONLY_IF_DIFFERENT)
file(COPY_FILE "${CMAKE_CURRENT_LIST_DIR}/two.c" "${AROS_SOURCE_DIR}/two.c" ONLY_IF_DIFFERENT)
include("${ENGINE_DIR}/SourceArchives.cmake")
set(one "${CMAKE_BINARY_DIR}/gen/objects/one.o")
set(two "${CMAKE_BINARY_DIR}/gen/objects/two.o")
aros_compile_literal_object(NAME one SOURCE "${AROS_SOURCE_DIR}/one.c" OUTPUT "${one}" LANGUAGE C)
aros_compile_literal_object(NAME two SOURCE "${AROS_SOURCE_DIR}/two.c" OUTPUT "${two}" LANGUAGE C)
set(objects "${two}" "${one}")
set(flags cr)
set(output "${AROS_DEVELOPER_LIB_DIR}/libsource.a")
if(PROBE STREQUAL "unregistered")
    list(APPEND objects "${CMAKE_BINARY_DIR}/gen/objects/unregistered.o")
elseif(PROBE STREQUAL "duplicate")
    list(APPEND objects "${one}")
elseif(PROBE STREQUAL "flags")
    set(flags "--plugin=unexpected")
elseif(PROBE STREQUAL "escape")
    set(output "${CMAKE_BINARY_DIR}/outside/libsource.a")
elseif(PROBE STREQUAL "collision")
    string(TOLOWER "${output}" folded)
    string(SHA256 key "${folded}")
    set_property(GLOBAL PROPERTY "AROS_LINKLIB_ARCHIVE_OWNER_${key}" previous)
elseif(PROBE STREQUAL "file-collision")
    string(TOLOWER "${output}" folded)
    string(SHA256 key "${folded}")
    set_property(GLOBAL PROPERTY "AROS_SDK_FILE_COPY_OUTPUT_${key}" previous)
elseif(PROBE STREQUAL "no-contract")
    set(AROS_NATIVE_BUILD_CONTRACT_VALIDATED OFF)
elseif(PROBE STREQUAL "remove-member")
    set(objects "${one}")
elseif(PROBE STREQUAL "interface-collision")
    add_library(source-archive-archive INTERFACE)
endif()
aros_source_archive(NAME source-archive OUTPUT "${output}" AR_FLAGS "${flags}" OBJECTS ${objects})
if(PROBE STREQUAL "link-consumer")
    add_executable(consumer "${CMAKE_CURRENT_LIST_DIR}/consumer.c")
    target_link_libraries(consumer PRIVATE source-archive-archive)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/selected-tools.cmake"
    "set(AR [==[${CMAKE_AR}]==])\nset(RANLIB [==[${CMAKE_RANLIB}]==])\n")
