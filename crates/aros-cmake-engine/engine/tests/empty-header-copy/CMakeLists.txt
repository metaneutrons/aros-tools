cmake_minimum_required(VERSION 3.22)
project(EmptyHeaderCopy NONE)
set(AROS_TARGET_CPU riscv)
set(AROS_TARGET_PLATFORM esp32p4)
set(AROS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(AROS_BUILD_DIR "${CMAKE_BINARY_DIR}")
set(AROS_NATIVE_BUILD_CONTRACT_VALIDATED TRUE)
# The production copy endpoint needs no compiler or broad SDK bootstrap.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bootstrap-stub")
file(WRITE "${CMAKE_BINARY_DIR}/bootstrap-stub/BootstrapSDK.cmake"
    "function(aros_bootstrap_sdk_includes)\nendfunction()\n")
list(INSERT CMAKE_MODULE_PATH 0 "${CMAKE_BINARY_DIR}/bootstrap-stub")
include("${AROS_ENGINE_DIR}/AROS.cmake")
if(PROBE MATCHES "^(copy-first|transform-first|empty-transform)$")
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/inputs")
    file(WRITE "${CMAKE_BINARY_DIR}/inputs/copied.h" "#define COPIED 1\n")
    file(WRITE "${CMAKE_BINARY_DIR}/inputs/input.h" "#define TRANSFORMED 1\n")
    if(PROBE STREQUAL "transform-first")
        aros_transform_header(NAME sample INPUT "${CMAKE_BINARY_DIR}/inputs/input.h"
            OUTPUT "${AROS_SDK_INCLUDE_DIR}/transformed.h" COPY_ONLY)
    endif()
    if(PROBE STREQUAL "empty-transform")
        aros_copy_includes(NAME sample DEST . SOURCE fixture PROVEN_EMPTY)
    else()
        aros_copy_includes(NAME sample DEST . SOURCE "${CMAKE_BINARY_DIR}/inputs"
            PATTERNS copied.h)
    endif()
    if(NOT PROBE STREQUAL "transform-first")
        aros_transform_header(NAME sample INPUT "${CMAKE_BINARY_DIR}/inputs/input.h"
            OUTPUT "${AROS_SDK_INCLUDE_DIR}/transformed.h" COPY_ONLY)
    endif()
elseif(PROBE STREQUAL "unrelated")
    add_custom_target(sample)
    aros_copy_includes(NAME sample DEST . SOURCE fixture PROVEN_EMPTY)
elseif(PROBE STREQUAL "nonempty")
    aros_copy_includes(NAME sample DEST . SOURCE fixture PROVEN_EMPTY PATTERNS header.h)
elseif(PROBE STREQUAL "unnamed")
    aros_copy_includes(DEST . SOURCE fixture PROVEN_EMPTY)
elseif(PROBE STREQUAL "excludes")
    aros_copy_includes(NAME sample DEST . SOURCE fixture PROVEN_EMPTY EXCLUDES header.h)
elseif(PROBE STREQUAL "override")
    aros_copy_includes(NAME sample DEST hidd SOURCE fixture PROVEN_EMPTY)
else()
    aros_copy_includes(NAME sample DEST . SOURCE fixture PROVEN_EMPTY)
    if(NOT TARGET sample)
        message(FATAL_ERROR "known empty source macro has no build endpoint")
    endif()
    aros_copy_includes(NAME unknown DEST . SOURCE fixture)
    if(TARGET unknown)
        message(FATAL_ERROR "missing patterns without proof created an endpoint")
    endif()
    aros_copy_includes(NAME foreign DEST . SOURCE arch/x86_64-pc/include PROVEN_EMPTY)
    if(TARGET foreign)
        message(FATAL_ERROR "foreign architecture created an empty endpoint")
    endif()
endif()
