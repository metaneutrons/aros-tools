cmake_minimum_required(VERSION 3.22)
project(RuntimeModuleOutputs C)
include("${CMAKE_CURRENT_LIST_DIR}/../../DefaultBuildClosure.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../RuntimeModuleOutputs.cmake")

file(WRITE "${CMAKE_BINARY_DIR}/selected.c" "int main(void) { return 0; }\n")
file(WRITE "${CMAKE_BINARY_DIR}/manual.c" "int main(void) { return 1; }\n")
file(WRITE "${CMAKE_BINARY_DIR}/unique.c" "int main(void) { return 2; }\n")
set(_directory "${CMAKE_BINARY_DIR}/SYS/Devs")
if(TEST_CASE STREQUAL reverse)
    set(_order manual-provider selected-provider)
else()
    set(_order selected-provider manual-provider)
endif()
foreach(_target IN LISTS _order)
    if(_target STREQUAL selected-provider)
        set(_source selected)
    else()
        set(_source manual)
    endif()
    add_executable("${_target}" "${CMAKE_BINARY_DIR}/${_source}.c")
    set(_name "example.resource")
    set(_output_dir "${_directory}")
    if(_target STREQUAL manual-provider AND TEST_CASE STREQUAL case-insensitive)
        set(_name "EXAMPLE.RESOURCE")
    elseif(_target STREQUAL manual-provider AND TEST_CASE STREQUAL separate-directory)
        set(_output_dir "${CMAKE_BINARY_DIR}/SYS/Libs")
    elseif(_target STREQUAL manual-provider AND TEST_CASE STREQUAL separate-suffix)
        set(_name "example.library")
    endif()
    set_target_properties("${_target}" PROPERTIES
        OUTPUT_NAME "${_name}" RUNTIME_OUTPUT_DIRECTORY "${_output_dir}")
    aros_register_runtime_module_output("${_target}" "${_output_dir}" "${_name}")
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${_target}.path"
        CONTENT "$<TARGET_FILE:${_target}>")
endforeach()

add_executable(unique-provider "${CMAKE_BINARY_DIR}/unique.c")
set_target_properties(unique-provider PROPERTIES
    OUTPUT_NAME unique.resource RUNTIME_OUTPUT_DIRECTORY "${_directory}")
aros_register_runtime_module_output(unique-provider "${_directory}" unique.resource)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/unique-provider.path"
    CONTENT "$<TARGET_FILE:unique-provider>")

if(NOT TEST_CASE STREQUAL no-root)
    add_custom_target(AROS)
    add_dependencies(AROS unique-provider)
    if(NOT TEST_CASE STREQUAL none-selected)
        add_dependencies(AROS selected-provider)
    endif()
    if(TEST_CASE STREQUAL both-selected OR
       TEST_CASE STREQUAL separate-directory OR TEST_CASE STREQUAL separate-suffix)
        add_dependencies(AROS manual-provider)
    endif()
    aros_limit_all_to_metamake_root(AROS)
    if(TEST_CASE STREQUAL late-conflict)
        # The final graph, not an earlier provisional selection, is authoritative.
        add_dependencies(AROS manual-provider)
        aros_limit_all_to_metamake_root(AROS)
    endif()
endif()
