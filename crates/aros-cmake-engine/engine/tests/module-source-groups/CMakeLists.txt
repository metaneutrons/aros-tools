cmake_minimum_required(VERSION 3.22)
project(ModuleSourceGroupsFixture LANGUAGES C CXX ASM)
if(NOT CASE STREQUAL disabled-objc)
    enable_language(OBJC)
endif()
if(NOT ENGINE_DIR)
    message(FATAL_ERROR "Explicit ENGINE_DIR required")
endif()
list(INSERT CMAKE_MODULE_PATH 0 "${CMAKE_CURRENT_LIST_DIR}/cmake")
set(AROS_TARGET_CPU riscv)
set(AROS_TARGET_PLATFORM fixture)
set(AROS_SOURCE_DIR "${CMAKE_BINARY_DIR}/source")
set(_generic "${AROS_SOURCE_DIR}/module")
set(_arch_a "${AROS_SOURCE_DIR}/arch/riscv-all/aaa")
set(_arch_z "${AROS_SOURCE_DIR}/arch/all-fixture/zzz")
file(MAKE_DIRECTORY "${_generic}" "${_arch_a}" "${_arch_z}" "${AROS_SOURCE_DIR}/compiler/startup")
foreach(_file IN ITEMS startup detach)
    file(WRITE "${AROS_SOURCE_DIR}/compiler/startup/${_file}.c" "int unused_${_file};\n")
endforeach()
foreach(_file IN ITEMS z first second)
    file(WRITE "${_generic}/${_file}.c" "int unused_${_file};\n")
endforeach()
file(WRITE "${_generic}/z.cpp" "int cxx_z;\n")
file(WRITE "${_generic}/language.m" "int objc_not_kobj;\n")
file(WRITE "${_generic}/assembler.S" ".text\n")
# Source-path sorting would select aaa/z before zzz/a. Make's architecture
# object wildcard selects a.o before z.o; the resolver must use that key.
file(WRITE "${_arch_a}/z.c" "int arch_z;\n")
file(WRITE "${_arch_z}/a.c" "int arch_a;\n")
include("${ENGINE_DIR}/AROS.cmake")
if(CASE STREQUAL missing-form)
elseif(CASE STREQUAL abi-only)
    aros_record_module_macro(OWNER fixture FORM abi-only)
elseif(CASE STREQUAL simple)
    aros_record_module_macro(OWNER fixture FORM simple)
else()
    aros_record_module_macro(OWNER fixture FORM full)
endif()
set(_arch_decls "riscv|arch/riscv-all/aaa|z" "fixture|arch/all-fixture/zzz|a")
set(_generic_sources second z first)
set(_assembly_sources assembler.S)
set(_cxx_sources z.cpp)
if(CASE STREQUAL simple)
    # These lanes are not prerequisites of the simple KOBJ macro.
    set(_assembly_sources absent.S)
    set(_cxx_sources absent.cpp)
endif()
if(CASE STREQUAL ambiguous-basename)
    file(WRITE "${_arch_a}/z.S" ".text\n")
    list(APPEND _arch_decls "riscv|arch/riscv-all/aaa|z.S")
elseif(CASE STREQUAL missing-arch)
    list(APPEND _arch_decls "riscv|arch/riscv-all/aaa|missing")
elseif(CASE STREQUAL missing-arch-directory)
    list(APPEND _arch_decls "riscv|arch/riscv-all/absent|missing")
elseif(CASE STREQUAL malformed-arch)
    list(APPEND _arch_decls "riscv|missing-third-field")
elseif(CASE STREQUAL missing-generic)
    list(APPEND _generic_sources absent)
elseif(CASE STREQUAL missing-assembly)
    list(APPEND _assembly_sources absent.S)
endif()
aros_resolve_kobj_source_groups(groups OWNER fixture DIRECTORY "${_generic}"
    SOURCES ${_generic_sources} CXX_SOURCES ${_cxx_sources} OBJC_SOURCES language.m
    ASM_SOURCES ${_assembly_sources} ARCH_SOURCES ${_arch_decls})
if(NOT "${groups_ARCH}" STREQUAL "${_arch_z}/a.c;${_arch_a}/z.c")
    message(FATAL_ERROR "Architecture group is not ordered by legacy object basename: ${groups_ARCH}")
endif()
if(NOT "${groups_C}" STREQUAL "${_generic}/second.c;${_generic}/first.c")
    message(FATAL_ERROR "C override or declaration order lost: ${groups_C}")
endif()
if(NOT "${groups_OBJC_DECLARED}" STREQUAL language.m)
    message(FATAL_ERROR "Objective-C declaration was lost")
endif()
if(CASE STREQUAL disabled-objc)
    if(groups_OBJC_NON_KOBJ)
        message(FATAL_ERROR "Disabled Objective-C language acquired a compile source")
    endif()
elseif(NOT "${groups_OBJC_NON_KOBJ}" STREQUAL "${_generic}/language.m")
    message(FATAL_ERROR "Objective-C non-KOBJ lane was lost")
endif()
if(CASE STREQUAL simple)
    if(groups_ASM OR groups_CXX)
        message(FATAL_ERROR "Simple source macro acquired full-only groups")
    endif()
else()
    if(NOT "${groups_CXX}" STREQUAL "${_generic}/z.cpp" OR
       NOT "${groups_ASM}" STREQUAL "${_generic}/assembler.S")
        message(FATAL_ERROR "Full KOBJ must preserve raw CXXFILES and ASMFILES")
    endif()
endif()
# GNU Make basename removes only the last suffix. A dotted stem must still
# replace its matching generic C source and keep its own architecture key.
file(WRITE "${_generic}/part.v2.c" "int generic_dotted;\n")
file(WRITE "${_arch_a}/part.v2.c" "int arch_dotted;\n")
aros_resolve_kobj_source_groups(dotted OWNER fixture DIRECTORY "${_generic}"
    SOURCES part.v2 ARCH_SOURCES "riscv|arch/riscv-all/aaa|part.v2")
if(dotted_C OR NOT "${dotted_ARCH}" STREQUAL "${_arch_a}/part.v2.c")
    message(FATAL_ERROR "Dotted object basename changed source override semantics")
endif()
# Fetched proxies must use logical declaration stems, not SHA filenames.
set(AROS_PORTS_DIR "${AROS_SOURCE_DIR}/ports")
file(MAKE_DIRECTORY "${AROS_PORTS_DIR}/generic" "${AROS_PORTS_DIR}/arch")
aros_resolve_sources(_generic_proxy "${AROS_PORTS_DIR}/generic"
    LANGUAGE C SOURCES z)
aros_resolve_sources(_arch_proxy_a "${AROS_PORTS_DIR}/arch" SOURCES a.c)
aros_resolve_sources(_arch_proxy_z "${AROS_PORTS_DIR}/arch" SOURCES z.c)
aros_resolve_kobj_source_groups(fetched OWNER fixture DIRECTORY "${_generic}"
    SOURCES "${AROS_PORTS_DIR}/generic/z"
    ARCH_SOURCES "riscv|ports/arch|z.c,a.c")
if(fetched_C OR NOT "${fetched_ARCH}" STREQUAL "${_arch_proxy_a};${_arch_proxy_z}")
    message(FATAL_ERROR "Fetched logical source stems did not preserve override/order: ${fetched_C}, ${fetched_ARCH}")
endif()
set(AROS_NATIVE_BUILD_CONTRACT fixture-source-contract)
set_property(GLOBAL PROPERTY AROS_MODULE_START_fixture "${CMAKE_BINARY_DIR}/generated/start.c")
set_property(GLOBAL PROPERTY AROS_MODULE_END_fixture "${CMAKE_BINARY_DIR}/generated/end.c")
aros_record_module_kobj_sources(OWNER fixture DIRECTORY "${_generic}"
    SOURCES second z first CXX_SOURCES ${_cxx_sources} OBJC_SOURCES language.m
    ASM_SOURCES ${_assembly_sources} ARCH_SOURCES ${_arch_decls})
foreach(_group IN ITEMS ARCH C CXX ASM OBJC_NON_KOBJ OBJC_DECLARED)
    get_property(_value GLOBAL PROPERTY "AROS_KOBJ_SOURCES_fixture_${_group}")
    if(NOT "${_value}" STREQUAL "${groups_${_group}}")
        message(FATAL_ERROR "Recorded ${_group} group differs")
    endif()
endforeach()
foreach(_group IN ITEMS START END)
    get_property(_value GLOBAL PROPERTY "AROS_KOBJ_SOURCES_fixture_${_group}")
    string(TOLOWER "${_group}" _lower)
    if(CASE STREQUAL simple)
        if(_value)
            message(FATAL_ERROR "Simple macro acquired generated ${_group}")
        endif()
    elseif(NOT "${_value}" STREQUAL "${CMAKE_BINARY_DIR}/generated/${_lower}.c")
        message(FATAL_ERROR "Source-owned ${_group} product not retained")
    endif()
endforeach()
if(CASE STREQUAL duplicate)
    aros_record_module_kobj_sources(OWNER fixture DIRECTORY "${_generic}" SOURCES first)
endif()
# The compile-group helper must clone the final target state and generator
# barrier, without depending on the runtime executable's own link.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/start.c" "int fixture_start;\n")
file(WRITE "${CMAKE_BINARY_DIR}/generated/end.c" "int fixture_end;\n")
add_executable(fixture EXCLUDE_FROM_ALL ${groups_ARCH} ${groups_C})
# Deferred quote includes are read through a consuming-target expression in
# each source's options. Merely copying COMPILE_OPTIONS loses that state.
set(_arch_include "${CMAKE_BINARY_DIR}/arch-include")
set(_fallback_include "${CMAKE_BINARY_DIR}/fallback-include")
file(MAKE_DIRECTORY "${_arch_include}" "${_fallback_include}")
file(WRITE "${_arch_include}/arch-only.h" "#define ARCH_HEADER 7\n")
file(WRITE "${_fallback_include}/fallback-only.h" "#define FALLBACK_HEADER 9\n")
file(WRITE "${_arch_a}/z.c" "#include \"arch-only.h\"\n#include \"fallback-only.h\"\nint arch_z = ARCH_HEADER + FALLBACK_HEADER;\n")
set_source_files_properties("${_arch_a}/z.c" PROPERTIES
    AROS_ARCH_QUOTE_INCLUDE "${_arch_include}")
aros_add_quote_options(fixture "-iquote${_fallback_include}")
set(_interface_include "${CMAKE_BINARY_DIR}/interface-include")
file(MAKE_DIRECTORY "${_interface_include}")
file(WRITE "${_interface_include}/interface-only.h" "#define INTERFACE_HEADER 11\n")
file(WRITE "${_generic}/first.c" "#include <interface-only.h>\nint unused_first = INTERFACE_HEADER;\n")
add_library(fixture-compile-interface INTERFACE)
target_include_directories(fixture-compile-interface INTERFACE "${_interface_include}")
target_link_libraries(fixture PRIVATE fixture-compile-interface)
target_compile_definitions(fixture PRIVATE KOBJ_TARGET_STATE=37)
target_compile_options(fixture PRIVATE -fno-common)
set_property(TARGET fixture PROPERTY C_STANDARD 11)
set_property(TARGET fixture PROPERTY AROS_NO_POSIXC_HEADERS TRUE)
add_custom_target(group-barrier COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_BINARY_DIR}/barrier.marker")
add_dependencies(fixture group-barrier)
if(CASE STREQUAL foreign)
    set_property(TARGET fixture PROPERTY AROS_FOREIGN_ARCH TRUE)
elseif(CASE STREQUAL private-runtime)
    set_property(TARGET fixture PROPERTY AROS_PRIVATE_RUNTIME_ALTERNATIVE TRUE)
elseif(CASE STREQUAL conflicting-group)
    add_library(fixture-kobj-arch OBJECT "${_generic}/first.c")
endif()
aros_create_kobj_compile_groups(objects fixture)
add_custom_target(compiled-groups DEPENDS ${objects_TARGETS})
get_target_property(_expected_options fixture COMPILE_OPTIONS)
foreach(_target IN LISTS objects_TARGETS)
    get_target_property(_defines "${_target}" COMPILE_DEFINITIONS)
    get_target_property(_options "${_target}" COMPILE_OPTIONS)
    get_target_property(_standard "${_target}" C_STANDARD)
    get_target_property(_deps "${_target}" MANUALLY_ADDED_DEPENDENCIES)
    get_target_property(_no_posix "${_target}" AROS_NO_POSIXC_HEADERS)
    get_target_property(_fallback "${_target}" AROS_QUOTE_FALLBACK_OPTIONS)
    if(NOT "${_defines}" STREQUAL KOBJ_TARGET_STATE=37 OR
       NOT "${_options}" STREQUAL "${_expected_options}" OR
       NOT "${_standard}" STREQUAL 11 OR NOT group-barrier IN_LIST _deps OR
       NOT _no_posix OR NOT "${_fallback}" STREQUAL "-iquote${_fallback_include}")
        message(FATAL_ERROR "${_target}: module compile state or barrier was not cloned: defines=${_defines}, options=${_options}, standard=${_standard}, deps=${_deps}")
    endif()
endforeach()
# Idempotent reuse must retain exact source order and ownership.
aros_create_kobj_compile_groups(reused fixture)
if(NOT "${objects_TARGETS}" STREQUAL "${reused_TARGETS}")
    message(FATAL_ERROR "Reusing source groups changed the target identities")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/arch-object-order.txt" CONTENT "$<JOIN:${objects_ARCH},\n>\n")
message(STATUS "Source group resolution and registration passed: ${CASE}")
